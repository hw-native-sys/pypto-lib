233: 
234:     The physical page allocator hands out one page per logical page of the
235:     longest request; the same table addresses the window, compressed and index
236:     caches of every source, so the returned page counts size all three pools.
237:     ``state_block_table`` assigns one compressor-state block per request.
238:     """
239:     if not requests:
240:         raise ValueError("a request plan needs at least one request")
241:     if any(length <= 0 or start < 0 for start, length in requests):
242:         raise ValueError("request positions must be non-negative and lengths positive")
243:     max_position = max(start + length - 1 for start, length in requests)
244:     table_width = (max_position + 1 + 127) // 128
245:     batch = len(requests)
246:     table = torch.full((batch, table_width), -1, dtype=torch.int32)
247:     next_page = 0
248:     positions = []
249:     request_rows = []
250:     for index, (start, length) in enumerate(requests):
251:         pages = (start + length + 127) // 128
252:         table[index, :pages] = torch.arange(next_page, next_page + pages, dtype=torch.int32)
253:         next_page += pages

255:         request_rows.append(torch.full((length,), index, dtype=torch.int32))
256:     positions = torch.cat(positions)
257:     request_ids = torch.cat(request_rows)
